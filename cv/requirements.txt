#   pip install --upgrade pip

#   (torchvision 0.22.x on PyPI requires torch 2.7.x; torchvision 0.23.x is the 2.8 line — keep them in one install.)
#   pip install -r cv/requirements.txt

# Official selector: https://pytorch.org/get-started/locally/ (Linux + CUDA 12.8)
#
# MediaPipe pose: set CV_POSE_BACKEND=auto|legacy|tasks|none (default auto). RunPod/Linux usually supports

#   apt-get update && apt-get install -y libgles2-mesa
